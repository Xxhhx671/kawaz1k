-- =============================================================================
-- Extrapolation
-- =============================================================================
local extrapEnabled = false
local extrapValue = 0.15

local function getExtrapolatedPos(part)
    if not part then return Vector3.zero end
    if not extrapEnabled then return part.Position end
    local vel = part.AssemblyLinearVelocity
    if not vel then return part.Position end
    if vel.Magnitude > 200 then return part.Position end
    return part.Position + vel * extrapValue
end

-- ===== Double / Triple Jump =====
local jumpCount = 0
local function handleMultiJump()
    clearConns("MultiJump")
    addC("MultiJump", UserInputService.JumpRequest:Connect(function()
        local hum = getHum()
        local hrp = getHRP()
        if not hum or not hrp then return end

        local maxJumps = 1
        if getT("TripleJump") then
            maxJumps = 3
        elseif getT("DoubleJump") then
            maxJumps = 2
        end

        if maxJumps > 1 and jumpCount < maxJumps and hum.FloorMaterial == Enum.Material.Air then
            jumpCount = jumpCount + 1
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
            hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, 40, hrp.AssemblyLinearVelocity.Z)
        end
    end))

    addC("MultiJumpReset", RunService.RenderStepped:Connect(function()
        local hum = getHum()
        if hum and hum.FloorMaterial ~= Enum.Material.Air then
            jumpCount = 0
        end
    end))
end

Features.DoubleJump = {
    on = function() handleMultiJump() end,
    off = function()

            clearConns("MultiJump"); clearConns("MultiJumpReset")

    end
}

Features.Fly = {
    on = function()
        local hrp = getHRP()
        if not hrp then setT("Fly", false); return end
        local speed = getV("fly_value", 60)
        local flyBV = Instance.new("BodyVelocity", hrp)
        flyBV.MaxForce = Vector3.new(1e6, 1e6, 1e6); flyBV.Velocity = Vector3.zero
        local flyBG = Instance.new("BodyGyro", hrp)
        flyBG.MaxTorque = Vector3.new(1e6, 1e6, 1e6); flyBG.P = 5000; flyBG.D = 500
        local h = getHum()
        if h then h.PlatformStand = true; h:ChangeState(Enum.HumanoidStateType.Physics) end

        addC("Fly", LocalPlayer.CharacterAdded:Connect(function()
            task.wait(0.3)
            if getT("Fly") then Features.Fly.off(); Features.Fly.on() end
        end))

        addC("Fly", RunService.RenderStepped:Connect(function()
            if not getT("Fly") then return end
            local r = getHRP()
            if not r or not flyBV or not flyBG then return end
            local cam = Camera
            local dir = Vector3.zero
            if UserInputService:GetFocusedTextBox() == nil then
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
            end
            local target = (dir.Magnitude > 0) and (dir.Unit * speed) or Vector3.zero
            flyBV.Velocity = flyBV.Velocity:Lerp(target, 0.25)
            flyBG.CFrame = cam.CFrame
        end))
    end,
    off = function()
        clearConns("Fly")
        local hrp = getHRP()
        if hrp then
            for _, v in ipairs(hrp:GetChildren()) do
                if v:IsA("BodyVelocity") or v:IsA("BodyGyro") then v:Destroy() end
            end
        end
        local h = getHum()
        if h then h.PlatformStand = false; h:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end
}

-- =============================================================================
-- WallCheck (kawaz 1:1)
-- =============================================================================
local function isPartVisible(part, char)
    if not part or not part.Parent then return false end
    local localChar = LocalPlayer.Character
    if not localChar then return false end
    local headPart = localChar:FindFirstChild("Head") or localChar:FindFirstChild("HumanoidRootPart")
    if not headPart then return false end

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {localChar, Camera, char}
    raycastParams.IgnoreWater = true

    local result = workspace:Raycast(headPart.Position, part.Position - headPart.Position, raycastParams)
    if not result then return true end
    if result.Instance == part then return true end
    if result.Instance:IsDescendantOf(part.Parent) then return true end
    return false
end

local espFolder = Instance.new("Folder", workspace)
espFolder.Name = "NursultanESP"
local espData = {}

local function clearESPFor(player)
    local d = espData[player]
    if d then
        if d.Highlight then d.Highlight:Destroy() end
        if d.InfoGui then d.InfoGui:Destroy() end
        if d.Box then d.Box:Destroy() end
        if d.DeathConn then d.DeathConn:Disconnect() end
    end
    espData[player] = nil
end

local function makeESPFor(player)
    if player == LocalPlayer or espData[player] then return end
    local data = {}
    espData[player] = data

    if getT("ESP_Highlight") then
        local hl = Instance.new("Highlight", espFolder)
        hl.FillColor = THEME.VisHidden
        hl.OutlineColor = THEME.AccentSoft
        hl.FillTransparency = 0.4
        hl.OutlineTransparency = 0
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        data.Highlight = hl
    end

    if getT("ESP_Name") or getT("ESP_Health") or getT("ESP_Distance") then
        local bb = Instance.new("BillboardGui", espFolder)
        bb.Name = "NursultanInfo"; bb.Size = UDim2.new(0, 160, 0, 50)
        bb.StudsOffset = Vector3.new(0, 3.2, 0); bb.AlwaysOnTop = true
        data.InfoGui = bb

        local nameLbl = Instance.new("TextLabel", bb)
        nameLbl.Size = UDim2.new(1, 0, 0, 16); nameLbl.BackgroundTransparency = 1
        nameLbl.Font = Enum.Font.GothamBold; nameLbl.TextSize = 12; nameLbl.TextColor3 = Color3.new(1, 1, 1)
        nameLbl.TextStrokeTransparency = 0.5; nameLbl.Text = player.Name; nameLbl.Visible = getT("ESP_Name")

        local distLbl = Instance.new("TextLabel", bb)
        distLbl.Size = UDim2.new(1, 0, 0, 12); distLbl.Position = UDim2.new(0, 0, 0, 16)
        distLbl.BackgroundTransparency = 1; distLbl.Font = Enum.Font.Gotham
        distLbl.TextSize = 10; distLbl.TextColor3 = Color3.new(1, 1, 1)
        distLbl.TextStrokeTransparency = 0.5; distLbl.Visible = getT("ESP_Distance")

        local hpBg = Instance.new("Frame", bb)
        hpBg.Size = UDim2.new(1, 0, 0, 4); hpBg.Position = UDim2.new(0, 0, 0, 30)
        hpBg.BackgroundColor3 = Color3.fromRGB(30, 30, 30); hpBg.BorderSizePixel = 0
        hpBg.Visible = getT("ESP_Health")
        Instance.new("UICorner", hpBg).CornerRadius = UDim.new(1, 0)

        local hp = Instance.new("Frame", hpBg)
        hp.Size = UDim2.new(1, 0, 1, 0); hp.BackgroundColor3 = THEME.Success; hp.BorderSizePixel = 0
        Instance.new("UICorner", hp).CornerRadius = UDim.new(1, 0)

        data.NameLbl = nameLbl; data.DistLbl = distLbl
        data.HealthBar = hp; data.HealthBg = hpBg
    end

    if getT("ESP_Box") then
        local box = Instance.new("Frame", ScreenGui)
        box.Name = "NursultanBox"; box.BackgroundTransparency = 1
        box.Visible = false; box.ZIndex = 5
        Instance.new("UICorner", box).CornerRadius = UDim.new(0, 6)
        local s = Instance.new("UIStroke", box)
        s.Color = THEME.VisHidden; s.Thickness = 1.5
        data.Box = box; data.BoxStroke = s
    end

    local function setupCharacter(char)
        local hrp = char:WaitForChild("HumanoidRootPart", 5)
        local hum = char:WaitForChild("Humanoid", 5)
        if data.Highlight then data.Highlight.Adornee = char end
        if data.InfoGui and hrp then data.InfoGui.Adornee = hrp end

        if hum then
            data.DeathConn = hum.Died:Connect(function()
                clearESPFor(player)
            end)
        end
    end

    if player.Character then setupCharacter(player.Character) end
    player.CharacterAdded:Connect(setupCharacter)
end

local function rebuildESP()
    for p, _ in pairs(espData) do clearESPFor(p) end
    for _, p in ipairs(Players:GetPlayers()) do makeESPFor(p) end
end

-- ESP цикл
RunService.RenderStepped:Connect(function()
    local myHrp = getHRP()
    for p, d in pairs(espData) do
        local char = p.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hrp or not head or not hum or hum.Health <= 0 then
            if d.InfoGui then d.InfoGui.Enabled = false end
            if d.Box then d.Box.Visible = false end
        else
            if d.InfoGui then d.InfoGui.Enabled = true end

            local visible = isPartVisible(head, char)
            local targetCol = getT("ESP_VisColor") and (visible and THEME.VisVisible or THEME.VisHidden) or THEME.Accent

            if d.Highlight then d.Highlight.FillColor = targetCol end
            if d.BoxStroke then d.BoxStroke.Color = targetCol end

            if d.DistLbl and myHrp and d.DistLbl.Visible then
                d.DistLbl.Text = string.format("[%dm]", math.floor((myHrp.Position - hrp.Position).Magnitude))
            end
            if d.HealthBar and d.HealthBg and d.HealthBg.Visible then
                local r = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                d.HealthBar.Size = UDim2.new(r, 0, 1, 0)
            end

            if d.Box and getT("ESP_Box") then
                local headPos, headOnScreen = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.6, 0))
                local footPos, footOnScreen = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))

                if headOnScreen and footOnScreen then
                    local height = math.abs(footPos.Y - headPos.Y)

local function isTeammate(plr)
    if not aimConfig.aim_team then return false end
    return plr.Team ~= nil and LocalPlayer.Team ~= nil and plr.Team == LocalPlayer.Team
end

local function getBestTargetPart(char)
    local selectedParts = getV("aim_selected_parts", {Head = true, Torso = true})
    local priorityChoice = getV("aim_priority_part", "Head")
    local checkWalls = aimConfig.aim_walls

    local candidateParts = {}

    if selectedParts["Head"] then table.insert(candidateParts, {name = "Head", part = char:FindFirstChild("Head")}) end
    if selectedParts["Torso"] then table.insert(candidateParts, {name = "Torso", part = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")}) end
    if selectedParts["Stomach"] then table.insert(candidateParts, {name = "Stomach", part = char:FindFirstChild("LowerTorso") or char:FindFirstChild("HumanoidRootPart")}) end
    if selectedParts["Arms"] then
        table.insert(candidateParts, {name = "Arms", part = char:FindFirstChild("LeftHand") or char:FindFirstChild("LeftArm")})
        table.insert(candidateParts, {name = "Arms", part = char:FindFirstChild("RightHand") or char:FindFirstChild("RightArm")})
    end
    if selectedParts["Legs"] then
        table.insert(candidateParts, {name = "Legs", part = char:FindFirstChild("LeftFoot") or char:FindFirstChild("LeftLeg")})
        table.insert(candidateParts, {name = "Legs", part = char:FindFirstChild("RightFoot") or char:FindFirstChild("RightLeg")})
    end

    table.sort(candidateParts, function(a, b)
        if a.name == priorityChoice and b.name ~= priorityChoice then
            return true
        elseif a.name ~= priorityChoice and b.name == priorityChoice then
            return false
        end
        return false
    end)

    for _, info in ipairs(candidateParts) do
        local p = info.part
        if p and (not checkWalls or isPartVisible(p, char)) then
            return p
        end
    end

    if not checkWalls then
        for _, info in ipairs(candidateParts) do
            if info.part then return info.part end
        end
    end

    return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local function findBestTargetSnap()
    local bestPart, bestScore = nil, math.huge
    local screenCenter = Camera.ViewportSize / 2
    local myHrp = getHRP()
    if not myHrp then return nil end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and not isTeammate(plr) then
            local char = plr.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if char and hum and hum.Health > 0 then
                local targetPartObj = getBestTargetPart(char)
                if targetPartObj then
                    local targetPos = getExtrapolatedPos(targetPartObj)
                    local sp, onScreen = Camera:WorldToViewportPoint(targetPos)
                    if onScreen then
                        local distToCenter = (Vector2.new(sp.X, sp.Y) - screenCenter).Magnitude
                        local worldDist = (targetPos - myHrp.Position).Magnitude
                        if distToCenter < aimConfig.aim_fov and worldDist < aimConfig.aim_max_dist and (not aimConfig.aim_walls or isPartVisible(targetPartObj, char)) then
                            if distToCenter < bestScore then
                                bestPart = targetPartObj; bestScore = distToCenter
                            end
                        end
                    end
                end
            end
        end
    end
    return bestPart
end

task.spawn(function()
    while true do
        task.wait(0.04)
        if getT("TriggerBot_Enabled") then
            local localChar = LocalPlayer.Character
            local tool = localChar and localChar:FindFirstChildOfClass("Tool")
            local mousePos = Camera.ViewportSize / 2
            local unitRay = Camera:ViewportPointToRay(mousePos.X, mousePos.Y)
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = {localChar, Camera}
            params.IgnoreWater = true

            local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * 1500, params)
            local triggered = false

            if result and result.Instance then
                local hitChar = result.Instance.Parent
                local enemyHum = hitChar and hitChar:FindFirstChildOfClass("Humanoid")
                if enemyHum and enemyHum.Health > 0 then
                    local enemyPlayer = Players:GetPlayerFromCharacter(hitChar)
                    if enemyPlayer and enemyPlayer ~= LocalPlayer and not isTeammate(enemyPlayer) then
                        triggered = true
                    end
                end
            end

            if triggered then
                pcall(function()
                    if tool then
                        tool:Activate()
                    else
                        local vu = game:GetService("VirtualUser")
                        vu:Button1Down(Vector2.zero)
                        task.wait(0.02)
                        vu:Button1Up(Vector2.zero)
                    end
                end)
                task.wait(0.1)
            end
        end
    end
end)

local function addSlider(parent, name, minV, maxV, defV, id, cb)
    local row = Instance.new("Frame", parent)
    row.Size = UDim2.new(1, 0, 0, 32); row.BackgroundTransparency = 1; row.ZIndex = 106

    local t = Instance.new("TextLabel", row)
    t.Size = UDim2.new(1, -45, 0, 13); t.BackgroundTransparency = 1
    t.Text = name; t.Font = Enum.Font.GothamMedium; t.TextSize = 10.5
    t.TextColor3 = THEME.TextPrimary; t.TextXAlignment = Enum.TextXAlignment.Left; t.ZIndex = 106

    local v = Instance.new("TextLabel", row)
    v.Size = UDim2.new(0, 35, 0, 13); v.Position = UDim2.new(1, -35, 0, 0)
    v.BackgroundTransparency = 1; v.Text = tostring(defV)
    v.Font = Enum.Font.GothamBold; v.TextSize = 9.5
    v.TextColor3 = THEME.Accent; v.TextXAlignment = Enum.TextXAlignment.Right; v.ZIndex = 106

    local track = Instance.new("Frame", row)
    track.Size = UDim2.new(1, 0, 0, 8); track.Position = UDim2.new(0, 0, 0, 20)
    track.BackgroundColor3 = THEME.ToggleOff; track.BorderSizePixel = 0; track.ZIndex = 106
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame", track)
    fill.Size = UDim2.new((defV - minV) / (maxV - minV), 0, 1, 0)
    fill.BackgroundColor3 = THEME.Accent; fill.BorderSizePixel = 0; fill.ZIndex = 107
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local function setVal(nv)
        nv = math.clamp(math.floor(nv + 0.5), minV, maxV)
        v.Text = tostring(nv)
        fill.Size = UDim2.new((nv - minV) / (maxV - minV), 0, 1, 0)
        setV(id, nv); if cb then cb(nv) end
    end

    local dragging = false
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setVal(minV + math.clamp((i.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            setVal(minV + math.clamp((i.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    setV(id, defV)
end

    isAnimating = true

    if open then
        MainContainer.Visible = true
        BlockerFrame.Visible = true
        setMenuBlur(true)
        Main.Size = UDim2.new(0, MENU_W - 80, 0, MENU_H - 50)

        local tw1 = TweenService:Create(MainContainer, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {GroupTransparency = 0})
        local tw2 = TweenService:Create(Main, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = UDim2.new(0, MENU_W, 0, MENU_H)})
        tw1:Play(); tw2:Play()
        tw2.Completed:Wait()
        isOpen = true
        isAnimating = false
    else
        isOpen = false
        setMenuBlur(false)

        local tw1 = TweenService:Create(MainContainer, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {GroupTransparency = 1})
        local tw2 = TweenService:Create(Main, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Size = UDim2.new(0, MENU_W - 80, 0, MENU_H - 50)})
        tw1:Play(); tw2:Play()
        tw2.Completed:Wait()

        MainContainer.Visible = false
        BlockerFrame.Visible = false
        isAnimating = false
    end
end
